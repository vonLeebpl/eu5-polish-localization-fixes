country_name_construction = {
	type = country

	# =========================================================================
	# SEKRETY ARCHITEKTURY - SEKCJA 1: ORYGINALNE WYJĄTKI STARTOWE NA SAMYM SZCZYCIE
	# =========================================================================
	text = {
		localization_key = country_name_construction_ROM_republic # Senatus Populusque Romanus
		trigger = {
			tag = ROM
			government_type = government_type:republic

			OR = {
				court_language ?= language:latin_language
				culture.language = language:latin_language
			}
		}
	}
	text = {
		localization_key = country_name_construction_BYZ_greek # Basileía Rhōmaíōn
		trigger = {
			has_game_rule = byz_name_greek

			OR = {
				tag = BYZ
				tag = ROM
			}

			country_rank = country_rank:rank_empire
			government_type = government_type:monarchy

			OR = {
				court_language ?= language:greek_language
				culture.language = language:greek_language
			}
		}
	}
	text = {
		localization_key = country_name_construction_name
		trigger = {
			OR = {
				AND = {
					tag = PAP
					has_reform = government_reform:papacy_reform
				}
				# WYJĄTEK PRIORYTETOWY: Saksonia-Lauenburg omija dynamiczne sita
				tag = LAU
			}
		}
	}
	
	# UNIKALNE KORONY DYNASTYCZNE (Kastylia, Aragonia, Czechy, unia Polski)
	text = {
		localization_key = country_name_construction_crown
		trigger = {
			country_rank_is_kingdom = yes
			OR = {
				tag = ARA
				tag = CAS
				AND = {
					tag = POL
					OR = {
						is_in_union = yes
						is_junior_partner = yes
					}
				}
				AND = {
					tag = BOH
					has_variable = crown_of_bohemia_variable
				}
			}
		}
	}

	# Cesarstwo chińskie: "Wielka Dynastia $NAME$" (Yuán -> Míng bez zmiany szyku)
	text = {
		localization_key = country_name_construction_pol_china_empire
		trigger = {
					country_rank ?= country_rank:rank_empire
					court_language ?= { language_family ?= language_family:chinese_language_family }
			}
	}

	# SPECYFICZNE UNIE I REFORMY PARADOXU (Przeniesione na szczyt dla bezbłędnej detekcji GBR/MAM/DAU/PAL)
	text = {
		localization_key = country_name_construction_prefix_rank_of_name_but_map_is_just_prefix_rank
		trigger = {
			OR = {
				AND = {
					tag = GBR
					coa_def_GBR_uk_trigger = yes
				}
				AND = {
					tag = MAM
					OR = {
						country_rank = country_rank:rank_empire
						country_rank = country_rank:rank_kingdom
					}
					religion.group = religion_group:muslim
				}
				AND = {
					tag = DAU
					OR = {
						country_rank = country_rank:rank_duchy
						country_rank = country_rank:rank_county
					}
				}
				AND = {
					tag = PAL
					OR = {
						country_rank = country_rank:rank_duchy
						country_rank = country_rank:rank_county
					}
				}
			}
		}
	}

	# =========================================================================
	# SEKCJA 2: NASZE WSPÓLNE POLSKIE DZIEŁO GRAMATYCZNE
	# =========================================================================
	
	# 1. SZTYWNY SZYK OPISOWY (Ograniczony - Wyjęto z niego republiki!)
	text = {
		localization_key = country_name_construction_pol_static_name
		trigger = { 
			OR = {
				tag = MOR
				tag = DLH
				government_type = government_type:tribe
				international_organization:hre ?= {
					country_has_special_status = {
						type = special_status:free_city
						country = ROOT
					}
				}
				# REFORMA: Całkowicie usunięto stąd AND = { government_type = republic ... }
				AND = {
					country_rank = country_rank:rank_county
					OR = {
						court_language ?= { language_family ?= language_family:indic_language_family }
						culture.language ?= { language_family ?= language_family:indic_language_family }
					}
				}
			}
		}
	}

	# 2. WYJĄTEK SPECJALNY: Zakon Szpitalników (Tag: KNI)
	text = {
		localization_key = country_name_construction_pol_knights_hospitaller
		trigger = { tag = KNI }
	}

	# 3. POZOSTAŁE ZAKONY RYCERSKIE (Szyk: Prefix + Ranga + Przymiotnik ki)
	text = {
		localization_key = country_name_construction_pol_military_order
		trigger = {
			has_reform = government_reform:military_order_reform
			NOT = { tag = KNI }
		}
	}

	# 3b. WYJĄTEK: Złota Orda (Tag: GLH) - $NAME$ zawiera już pełną nazwę "Złota Orda"
	text = {
		localization_key = country_name_construction_pol_golden_horde
		trigger = {
			tag = GLH
			government_type = government_type:steppe_horde
		}
	}

	# 4. ORDY (Koczownicy) - CAŁKOWICIE WYDZIELONE (Żeńska końcówka -ka)
	text = {
		localization_key = country_name_construction_pol_horde
		trigger = {
			government_type = government_type:steppe_horde
		}
	}

	# 4b. WYJĄTEK: Mali (tag MAL) - zawsze mianownik, mimo islamu
	text = {
		localization_key = country_name_construction_pol_far_east_descriptive
		trigger = {
			tag = MAL
		}
	}

	# 4c. WYJĄTEK: Tybet / Guge - forma przymiotnikowa ("Teokracja Tybetańska")
	text = {
		localization_key = country_name_construction_pol_tibet
		trigger = {
			OR = { tag = TIB tag = GUG }
		}
	}

	# 4d. RANGI INDYJSKIE (Thikana / Rāj / Mahārājya) - zawsze mianownik, także muzułmańskie.
	#     Warunek jak w country_ranks.txt: język (dworu lub kultury) indyjski/drawidyjski/malajski, ranga < cesarstwo.
	text = {
		localization_key = country_name_construction_pol_far_east_descriptive
		trigger = {
			NOT = { country_rank = country_rank:rank_empire }
			OR = {
				culture.language ?= { language_family ?= language_family:indic_language_family }
				culture.language ?= { language_family ?= language_family:dravidian_language_family }
				culture.language ?= language:malay_language
				court_language ?= { language_family ?= language_family:indic_language_family }
				court_language ?= { language_family ?= language_family:dravidian_language_family }
				court_language ?= language:malay_language
			}
		}
	}

	# 5. ŚWIAT ISLAMU (Sułtanaty i Emiraty) - Z WYKLUCZENIEM ORD!
	text = {
		localization_key = country_name_construction_pol_islamic_rank
		trigger = {
			religion.group = religion_group:muslim
			NOT = { government_type = government_type:steppe_horde }
		}
	}

	# 5b. NIE-EUROPA - szyk opisowy (mianownik). ODWROCONE: łapie WSZYSTKO poza rodzinami
	#     językowymi o utrwalonej polskiej przymiotnikowości. Wyjątek: khmerski (-> przymiotnik).
	#     Ordy/plemiona/muzułmanie/Chiny-cesarstwo/szogunat łapią wcześniejsze bloki.
	text = {
		localization_key = country_name_construction_pol_far_east_descriptive
		trigger = {
			# Zachodnia półkula (Ameryka Płn. i Pd. są zagnieżdżone pod 'america')
			# - bliżej reguł europejskich (przymiotnik), pomijamy.
			NOT = { capital.continent = continent:america }
			NOT = { court_language ?= language:khmer_language }
			NOT = { culture.language ?= language:khmer_language }
			NOT = { court_language ?= { language_family ?= language_family:germanic_language_family } }
			NOT = { court_language ?= { language_family ?= language_family:romance_language_family } }
			NOT = { court_language ?= { language_family ?= language_family:slavic_language_family } }
			NOT = { court_language ?= { language_family ?= language_family:celtic_language_family } }
			NOT = { court_language ?= { language_family ?= language_family:iranic_language_family } }
			NOT = { court_language ?= { language_family ?= language_family:uralic_language_family } }
			NOT = { court_language ?= { language_family ?= language_family:semitic_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:germanic_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:romance_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:slavic_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:celtic_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:iranic_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:uralic_language_family } }
			NOT = { culture.language ?= { language_family ?= language_family:semitic_language_family } }
			# Języki europejskie BEZ przypisanej rodziny (utrwalone polskie przymiotniki / lista dopełniaczowa)
			NOT = { court_language ?= language:greek_language }
			NOT = { court_language ?= language:albanian_language }
			NOT = { court_language ?= language:armenian_language }
			NOT = { court_language ?= language:georgian_language }
			NOT = { court_language ?= language:basque_language }
			NOT = { court_language ?= language:baltic_language }
			NOT = { culture.language ?= language:greek_language }
			NOT = { culture.language ?= language:albanian_language }
			NOT = { culture.language ?= language:armenian_language }
			NOT = { culture.language ?= language:georgian_language }
			NOT = { culture.language ?= language:basque_language }
			NOT = { culture.language ?= language:baltic_language }
		}
	}

	# 6. Europejskie Monarchie Biskupie Wyższe (Teokracje o randze Cesarstwa, Królestwa i Księstwa)
	text = {
		localization_key = country_name_construction_pol_bishopric_monarchy
		trigger = {
			government_type = government_type:theocracy
			religion.group = religion_group:christian
			OR = {
				country_rank = country_rank:rank_empire
				country_rank = country_rank:rank_kingdom
				country_rank = country_rank:rank_duchy
			}
		}
	}

	# 7. Europejskie Teokracje Niższe / Biskupstwa (Pozostała ranga Hrabstwa - County)

			government_type = government_type:theocracy
			religion.group = religion_group:christian
			country_rank = country_rank:rank_county
		}

		trigger = {
			government_type = government_type:republic
		}

		localization_key = country_name_construction_pol_empire_descriptive

		trigger = {

		}

	# 11. SITO 1: MONARCHIE EUROPEJSKIE W DOPEŁNIACZU (WYŻSZY PRIORYTET)
	# =========================================================================
	text = {
		localization_key = country_name_construction_pol_euro_monarchy_master
		trigger = {
			government_type = government_type:monarchy
			OR = {
				country_rank = country_rank:rank_kingdom
				country_rank = country_rank:rank_duchy
				country_rank = country_rank:rank_county
			}
			OR = {
				# Główne Monarchie, Bałkany i regiony rdzennie toponimiczne
				tag = FRA tag = ENG tag = ARA tag = CAS tag = NAV tag = POR tag = SCO tag = DAN tag = SWE tag = NOR tag = HUN tag = BOS tag = SER tag = BUL tag = WAL tag = MOL tag = ALB tag = HUM tag = ZET tag = CRO
				# Kraje Alpejskie, Rzesza i Krym (HAB dla Austrii, LBV - Dolna Bawaria, UBV - Górna Bawaria, THE - Teodoro, LAU - Saksonia-Lauenburg)
				tag = LBV tag = UBV tag = PAL tag = WUR tag = BAD tag = HAB tag = STY tag = CAR tag = TYR tag = SLZ tag = MKL tag = ANH tag = ERF tag = LAU tag = THE
				# Niderlandy i Dolna Nadrenia
				tag = LUX tag = BRB tag = FLA tag = HOL tag = GEL
				# Inne kraje europejskie, które nie pasują do powyższych kategorii
				tag = GEO tag = BOH
			}
		}
	}
	
	# =========================================================================
	# 12. SITO 2: MONARCHIE GLOBALNE, DYNAMICZNE I MARCHIE (NISKI PRIORYTET)
